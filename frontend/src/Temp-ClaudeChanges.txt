const express = require("express");
const bodyParser = require("body-parser");
const basicRoutes = require("./routes/basicRoutes");
const bodyRoutes = require("./routes/bodyRoutes");
const authRoutes = require("./routes/authRoutes");
const fileRoutes = require("./routes/fileRoutes");
const advancedRoutes = require("./routes/advancedRoutes");

const app = express();
const PORT = 3000;

// Middleware
app.use(bodyParser.json());
app.use(bodyParser.urlencoded({ extended: true }));

// CORS — must be before all routes
app.use((req, res, next) => {
    res.header("Access-Control-Allow-Origin", "*");
    res.header("Access-Control-Allow-Headers", "*");
    res.header("Access-Control-Allow-Methods", "GET, POST, PUT, PATCH, DELETE, OPTIONS");

    // Handle preflight — browser sends OPTIONS before POST/PUT/PATCH/DELETE
    // If you don't respond 204 here, it falls into your routes and gets a 404
    if (req.method === "OPTIONS") {
        return res.sendStatus(204);
    }

    next();
});

// Routes
app.use("/api", basicRoutes);
app.use("/api/body", bodyRoutes);
app.use("/api/auth", authRoutes);
app.use("/api/file", fileRoutes);
app.use("/api/advanced", advancedRoutes);

app.listen(PORT, () => {
    console.log(`🚀 Server running at http://localhost:${PORT}`);
});
